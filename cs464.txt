using System;
using System.Data.Entity;
using System.Linq;
using System.Windows;
using System.Windows.Controls;
using QUANLYSINHVIEN.Model;

namespace QUANLYSINHVIEN
{
    public partial class MainWindow : Window
    {
        private bool _dangTaiDuLieu;

        public MainWindow()
        {
            InitializeComponent();
        }

        // Câu 3: Tải khoa vào ComboBox và sinh viên vào DataGrid (Entity Framework Database First).
        private void Window_Loaded(object sender, RoutedEventArgs e)
        {
            try
            {
                TaiDuLieu();
                XoaTrangForm();
            }
            catch (Exception ex)
            {
                BaoLoi("Không thể kết nối CSDL. Hãy chạy file SQL và kiểm tra App.config.", ex);
            }
        }

        private void TaiDuLieu()
        {
            _dangTaiDuLieu = true;
            try
            {
                using (var db = new QLSINHVIENEntities())
                {
                    // AsNoTracking để có dữ liệu mới nhất và không lưu trạng thái chỉnh sửa của DataGrid.
                    cboKhoa.ItemsSource = db.KHOAs.AsNoTracking()
                        .OrderBy(k => k.TenKhoa).ToList();

                    // Include(KHOA) để hiển thị TenKhoa ngay cả sau khi đóng DbContext.
                    dgvSinhVien.ItemsSource = db.SINHVIENs.AsNoTracking()
                        .Include(s => s.KHOA)
                        .OrderBy(s => s.MaSV)
                        .ToList();
                }
            }
            finally
            {
                _dangTaiDuLieu = false;
            }
        }

        // Câu 4: Khi chọn một dòng, hiển thị đầy đủ dữ liệu lên các control nhập liệu.
        private void dgvSinhVien_SelectionChanged(object sender, SelectionChangedEventArgs e)
        {
            if (_dangTaiDuLieu) return;

            var sv = dgvSinhVien.SelectedItem as SINHVIEN;
            if (sv == null) return;

            txtMaSV.Text = sv.MaSV;
            txtMaSV.IsReadOnly = true; // Không cho đổi mã khóa chính khi sửa.
            txtHoTen.Text = sv.HoTen;
            dpNgaySinh.SelectedDate = sv.NgaySinh;
            rbNam.IsChecked = sv.GioiTinh == "Nam";
            rbNu.IsChecked = sv.GioiTinh == "Nữ";
            rbKhac.IsChecked = sv.GioiTinh == "Khác";
            cboKhoa.SelectedValue = sv.MaKhoa;
        }

        // Câu 4: Làm mới form, bỏ chọn và tải lại DataGrid.
        private void btnLamMoi_Click(object sender, RoutedEventArgs e)
        {
            try
            {
                TaiDuLieu();
                XoaTrangForm();
            }
            catch (Exception ex)
            {
                BaoLoi("Không tải lại được dữ liệu.", ex);
            }
        }

        private void XoaTrangForm()
        {
            dgvSinhVien.SelectedItem = null;
            txtMaSV.IsReadOnly = false;
            txtMaSV.Clear();
            txtHoTen.Clear();
            dpNgaySinh.SelectedDate = null;
            rbNam.IsChecked = false;
            rbNu.IsChecked = false;
            rbKhac.IsChecked = false;
            cboKhoa.SelectedIndex = -1;
            txtMaSV.Focus();
        }

        // Dùng chung cho Thêm và Sửa: kiểm tra dữ liệu nhập.
        private bool LayThongTinNhap(out string maSV, out string hoTen,
            out DateTime ngaySinh, out string gioiTinh, out string maKhoa)
        {
            maSV = txtMaSV.Text.Trim();
            hoTen = txtHoTen.Text.Trim();
            ngaySinh = DateTime.MinValue;
            gioiTinh = null;
            maKhoa = cboKhoa.SelectedValue as string;

            if (string.IsNullOrWhiteSpace(maSV))
            {
                MessageBox.Show("Vui lòng nhập mã sinh viên.");
                txtMaSV.Focus();
                return false;
            }
            if (string.IsNullOrWhiteSpace(hoTen))
            {
                MessageBox.Show("Vui lòng nhập họ tên.");
                txtHoTen.Focus();
                return false;
            }
            if (!dpNgaySinh.SelectedDate.HasValue)
            {
                MessageBox.Show("Vui lòng chọn ngày sinh.");
                return false;
            }
            ngaySinh = dpNgaySinh.SelectedDate.Value;

            if (rbNam.IsChecked == true) gioiTinh = "Nam";
            else if (rbNu.IsChecked == true) gioiTinh = "Nữ";
            else if (rbKhac.IsChecked == true) gioiTinh = "Khác";

            if (gioiTinh == null)
            {
                MessageBox.Show("Vui lòng chọn giới tính.");
                return false;
            }
            if (string.IsNullOrWhiteSpace(maKhoa))
            {
                MessageBox.Show("Vui lòng chọn khoa.");
                return false;
            }
            return true;
        }

        // Câu 5: THÊM - INSERT vào CSDL, tải lại DataGrid.
        private void btnThem_Click(object sender, RoutedEventArgs e)
        {
            if (!LayThongTinNhap(out string maSV, out string hoTen,
                    out DateTime ngaySinh, out string gioiTinh, out string maKhoa)) return;

            try
            {
                using (var db = new QLSINHVIENEntities())
                {
                    if (db.SINHVIENs.Any(s => s.MaSV == maSV))
                    {
                        MessageBox.Show("Mã sinh viên đã tồn tại.", "Thông báo",
                            MessageBoxButton.OK, MessageBoxImage.Warning);
                        return;
                    }

                    var sv = new SINHVIEN
                    {
                        MaSV = maSV,
                        HoTen = hoTen,
                        NgaySinh = ngaySinh,
                        GioiTinh = gioiTinh,
                        MaKhoa = maKhoa
                    };
                    db.SINHVIENs.Add(sv);
                    db.SaveChanges();
                }

                TaiDuLieu();
                XoaTrangForm();
                MessageBox.Show("Thêm sinh viên thành công!");
            }
            catch (Exception ex)
            {
                BaoLoi("Không thể thêm sinh viên.", ex);
            }
        }

        // Câu 5: SỬA - cập nhật sinh viên đang chọn, giữ nguyên khóa chính MaSV.
        private void btnSua_Click(object sender, RoutedEventArgs e)
        {
            var dangChon = dgvSinhVien.SelectedItem as SINHVIEN;
            if (dangChon == null)
            {
                MessageBox.Show("Hãy chọn sinh viên cần sửa trong DataGrid.");
                return;
            }
            if (!LayThongTinNhap(out string maSV, out string hoTen,
                    out DateTime ngaySinh, out string gioiTinh, out string maKhoa)) return;

            try
            {
                using (var db = new QLSINHVIENEntities())
                {
                    var sv = db.SINHVIENs.Find(dangChon.MaSV);
                    if (sv == null)
                    {
                        MessageBox.Show("Sinh viên không còn tồn tại trong CSDL.");
                        TaiDuLieu();
                        XoaTrangForm();
                        return;
                    }
                    sv.HoTen = hoTen;
                    sv.NgaySinh = ngaySinh;
                    sv.GioiTinh = gioiTinh;
                    sv.MaKhoa = maKhoa;
                    db.SaveChanges();
                }

                TaiDuLieu();
                XoaTrangForm();
                MessageBox.Show("Sửa sinh viên thành công!");
            }
            catch (Exception ex)
            {
                BaoLoi("Không thể sửa sinh viên.", ex);
            }
        }

        // Câu 5: XÓA - xóa sinh viên đang chọn, có hộp thoại xác nhận.
        private void btnXoa_Click(object sender, RoutedEventArgs e)
        {
            var dangChon = dgvSinhVien.SelectedItem as SINHVIEN;
            if (dangChon == null)
            {
                MessageBox.Show("Hãy chọn sinh viên cần xóa trong DataGrid.");
                return;
            }

            var traLoi = MessageBox.Show(
                $"Bạn có chắc muốn xóa sinh viên {dangChon.MaSV} - {dangChon.HoTen}?",
                "Xác nhận xóa", MessageBoxButton.YesNo, MessageBoxImage.Question);
            if (traLoi != MessageBoxResult.Yes) return;

            try
            {
                using (var db = new QLSINHVIENEntities())
                {
                    var sv = db.SINHVIENs.Find(dangChon.MaSV);
                    if (sv == null)
                    {
                        MessageBox.Show("Sinh viên không còn tồn tại trong CSDL.");
                    }
                    else
                    {
                        db.SINHVIENs.Remove(sv);
                        db.SaveChanges();
                        MessageBox.Show("Xóa sinh viên thành công!");
                    }
                }

                TaiDuLieu();
                XoaTrangForm();
            }
            catch (Exception ex)
            {
                BaoLoi("Không thể xóa sinh viên.", ex);
            }
        }

        private void BaoLoi(string thongBao, Exception ex)
        {
            // GetBaseException() thường trả về lỗi SQL Server chi tiết và dễ kiểm tra hơn.
            MessageBox.Show(thongBao + "\n\nChi tiết: " + ex.GetBaseException().Message,
                "Lỗi", MessageBoxButton.OK, MessageBoxImage.Error);
        }
    }
}

using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;
using System.Threading.Tasks;

namespace QLSV_C5
{
    /// <summary>
    /// Interaction logic for MainWindow.xaml
    /// </summary>
    public partial class MainWindow : Window
    {
        QLSVEntities db = new QLSVEntities();
        private SINHVIEN svDangChon;
        public MainWindow()
        {
            InitializeComponent();
            loadData();
        }
        private void loadData()
        {
            DG_SinhVien.ItemsSource = db.SINHVIENs.ToList();
            cbb_Mon.ItemsSource = db.MONHOCs.ToList();
            cbb_Mon.DisplayMemberPath = "TenMon";
            cbb_Mon.SelectedValuePath = "MaMon";
        }
        private string LayGioiTinh()
        {
            if (rdb_khac.IsChecked == true) return "Khác";
            if (rdb_nam.IsChecked == true) return "Nam";
            if (rdb_nu.IsChecked == true) return "Nữ";
            return "";
        }
        private DateTime LayNgaySinh()
        {
            if (dp_NgaySinh.SelectedDate == null)
                return DateTime.Now;
            else
                return dp_NgaySinh.SelectedDate.Value;
        }
        private string LayMonHoc()
        {
            if (cbb_Mon.SelectedValue != null)
                return cbb_Mon.SelectedValue.ToString();
            else
                return "";
        }
        private void ClearForm()
        {
            txt_MaSV.Clear();
            txt_HoTen.Clear();
            txt_DiaChi.Clear();
            rdb_nam.IsChecked = false;
            rdb_nu.IsChecked = false;
            rdb_khac.IsChecked = false;
            dp_NgaySinh.SelectedDate = null;
            cbb_Mon.SelectedIndex = -1;
            svDangChon = null;
        }

        private void Button_Them(object sender, RoutedEventArgs e)
        {
            SINHVIEN sv = new SINHVIEN
            {
                MaSV = txt_MaSV.Text.Trim(),
                HoTen = txt_HoTen.Text.Trim(),
                DiaChi = txt_DiaChi.Text.Trim(),
                GioiTinh = LayGioiTinh(),
                NgaySinh = LayNgaySinh(),
                MaMon = LayMonHoc()
            };
            db.SINHVIENs.Add(sv);
            db.SaveChanges();
            MessageBox.Show("Đã thêm sinh viên thành công,", "Thông báo",
                MessageBoxButton.OK, MessageBoxImage.Information);
            loadData();
            ClearForm();
        }

        private void Button_Sua(object sender, RoutedEventArgs e)
        {
            if (svDangChon == null)
            {
                MessageBox.Show("Vui lòng chọn sinh viên để sửa!", "Thông báo",
                    MessageBoxButton.OK, MessageBoxImage.Warning);
            }
            else{
                SINHVIEN sv = db.SINHVIENs.Find(svDangChon.MaSV);
                if(sv.MaSV != txt_MaSV.Text.Trim())
                   MessageBox.Show("Không thể sửa mã sinh viên", "Thông báo",
                    MessageBoxButton.OK, MessageBoxImage.Error);
                else{
                    sv.HoTen = txt_HoTen.Text.Trim();
                    sv.DiaChi = txt_DiaChi.Text.Trim();
                    sv.GioiTinh = LayGioiTinh();
                    sv.NgaySinh = LayNgaySinh();
                    sv.MaMon = LayMonHoc();
                    db.SaveChanges();
                    MessageBox.Show("Đã sửa sinh viên thành công!", "Thông báo",
                                        MessageBoxButton.OK, MessageBoxImage.Information);
                }
                loadData();ClearForm();
            }
        }

        private void Button_xoa(object sender, RoutedEventArgs e)
        {
            if (svDangChon == null)
                MessageBox.Show("Vui lòng chọn sinh viên để xóa!", "Thông báo",
                    MessageBoxButton.OK, MessageBoxImage.Warning);
            else
            {
                db.SINHVIENs.Remove(svDangChon);
                db.SaveChanges();
                MessageBox.Show("Đã xóa thành công!", "Thông báo",
                                    MessageBoxButton.OK, MessageBoxImage.Information);
                loadData();
                ClearForm();
            }
        }

        private void Button_Tim(object sender, RoutedEventArgs e)
        {
            string keyword = txt_tim.Text.Trim().ToLower();
            List<SINHVIEN> ketQua = new List<SINHVIEN>();
            if (keyword == ""){
                MessageBox.Show("Hãy nhập tên sinh viên để tìm", "Thông báo",MessageBoxButton.OK, MessageBoxImage.Warning);
            }
            else{
                foreach (SINHVIEN sv in db.SINHVIENs)
                {
                    if (sv.HoTen != null && sv.HoTen.ToLower().Contains(keyword)){
                        ketQua.Add(sv);
                    }
                }
                if (ketQua.Count > 0){
                    DG_SinhVien.ItemsSource = ketQua;
                }
                else{
                    MessageBox.Show("Không tìm thấy sinh viên nào!", "Thông báo",
                        MessageBoxButton.OK, MessageBoxImage.Information);
                }
            }
        }
        private void DG_SinhVien_SelectionChanged(object sender, SelectionChangedEventArgs e)
        {
            svDangChon = DG_SinhVien.SelectedItem as SINHVIEN;
            if (svDangChon != null)
            {
                txt_MaSV.Text = svDangChon.MaSV;
                txt_HoTen.Text = svDangChon.HoTen;
                txt_DiaChi.Text = svDangChon.DiaChi;
                if (svDangChon.GioiTinh == "Nam") rdb_nam.IsChecked = true;
                if (svDangChon.GioiTinh == "Nữ") rdb_nu.IsChecked = true;
                if (svDangChon.GioiTinh == "Khác") rdb_khac.IsChecked = true;
                dp_NgaySinh.SelectedDate = svDangChon.NgaySinh;
                cbb_Mon.SelectedValue = svDangChon.MaMon;
            }
        }
        private void Button_Load(object sender, RoutedEventArgs e)
        {
            loadData();
        }
    }
}

                if (s.MaTL == maTL)
                {
                    loc_CB.Add(s);
                }
            }
            dgSach.ItemsSource = loc_CB;

namespace QUANLYNHANVIEN

    /// <summary>
    /// De thi thu 1 - Quan ly nhan vien (WPF .NET Framework + EF6 Database First).
    /// Yeu cau Entity Data Model tao cac lop: QLNHANVIENEntities, NHANVIEN, PHONGBAN.
    /// Cac control XAML: txtMaNV, txtHoTen, dpNgaySinh, rbNam, rbNu,

    {
        private readonly QLNHANVIENEntities db = new QLNHANVIENEntities();

        public MainWindow()
        {
            InitializeComponent();
            TaiDuLieu();
        }

        // CAU 3: Tai danh sach phong ban va nhan vien.
        private void TaiDuLieu()
        {
            cboPhongBan.ItemsSource = db.PHONGBANs.ToList();
            cboPhongBan.DisplayMemberPath = "TenPB";
            cboPhongBan.SelectedValuePath = "MaPB";

            TaiDuLieuNhanVien();
        }

        private void TaiDuLieuNhanVien()
        {
            // Include de cot PHONGBAN.TenPB trong DataGrid hien thi dung.
            dgNhanVien.ItemsSource = db.NHANVIENs
                .Include("PHONGBAN")
                .ToList();
        }

        // CAU 4: Chon dong tren DataGrid -> dua du lieu len form.
        private void dgNhanVien_SelectionChanged(object sender, SelectionChangedEventArgs e)
        {
            NHANVIEN nv = dgNhanVien.SelectedItem as NHANVIEN;
            if (nv == null)
                return;

            txtMaNV.Text = nv.MaNV;
            txtHoTen.Text = nv.HoTen;
            dpNgaySinh.SelectedDate = nv.NgaySinh;
            cboPhongBan.SelectedValue = nv.MaPB;

            rbNam.IsChecked = nv.GioiTinh == "Nam";
            rbNu.IsChecked = nv.GioiTinh == "Nữ";
            rbKhac.IsChecked = nv.GioiTinh == "Khác";

            // MaNV la khoa chinh: khong cho doi khi sua.
            txtMaNV.IsReadOnly = true;
        }

        private string LayGioiTinh()
        {
            if (rbNam.IsChecked == true) return "Nam";
            if (rbNu.IsChecked == true) return "Nữ";
            if (rbKhac.IsChecked == true) return "Khác";
            return null;
        }

        private bool KiemTraDuLieu()
        {
            if (string.IsNullOrWhiteSpace(txtMaNV.Text) ||
                string.IsNullOrWhiteSpace(txtHoTen.Text))
            {
                MessageBox.Show("Vui lòng nhập mã nhân viên và họ tên.");
                return false;
            }

            if (!dpNgaySinh.SelectedDate.HasValue)
            {
                MessageBox.Show("Vui lòng chọn ngày sinh.");
                return false;
            }

            if (LayGioiTinh() == null)
            {
                MessageBox.Show("Vui lòng chọn giới tính.");
                return false;
            }

            if (cboPhongBan.SelectedValue == null)
            {
                MessageBox.Show("Vui lòng chọn phòng ban.");
                return false;
            }

            return true;
        }

        // CAU 5: Them nhan vien.
        private void btnThem_Click(object sender, RoutedEventArgs e)
        {
            if (!KiemTraDuLieu()) return;

            string maNV = txtMaNV.Text.Trim();
            if (db.NHANVIENs.Any(x => x.MaNV == maNV))
            {
                MessageBox.Show("Mã nhân viên đã tồn tại.");
                return;
            }

            NHANVIEN nv = new NHANVIEN
            {
                MaNV = maNV,
                HoTen = txtHoTen.Text.Trim(),
                NgaySinh = dpNgaySinh.SelectedDate.Value,
                GioiTinh = LayGioiTinh(),
                MaPB = cboPhongBan.SelectedValue.ToString()
            };

            try
            {
                db.NHANVIENs.Add(nv);
                db.SaveChanges();
                MessageBox.Show("Thêm nhân viên thành công!");
                LamMoiForm();
            }
            catch (Exception ex)
            {
                // Khong giu lai ban ghi them loi trong DbContext.
                if (db.Entry(nv).State == EntityState.Added)
                    db.Entry(nv).State = EntityState.Detached;
                MessageBox.Show("Không thể thêm nhân viên: " + ex.Message);
            }
        }

        // CAU 5: Sua nhan vien duoc chon.
        private void btnSua_Click(object sender, RoutedEventArgs e)
        {
            NHANVIEN dangChon = dgNhanVien.SelectedItem as NHANVIEN;
            if (dangChon == null)
            {
                MessageBox.Show("Vui lòng chọn nhân viên cần sửa.");
                return;
            }

            if (!KiemTraDuLieu()) return;

            try
            {
                NHANVIEN nv = db.NHANVIENs.Find(dangChon.MaNV);
                if (nv == null)
                {
                    MessageBox.Show("Nhân viên không còn tồn tại.");
                    LamMoiForm();
                    return;
                }

                nv.HoTen = txtHoTen.Text.Trim();
                nv.NgaySinh = dpNgaySinh.SelectedDate.Value;
                nv.GioiTinh = LayGioiTinh();
                nv.MaPB = cboPhongBan.SelectedValue.ToString();

                db.SaveChanges();
                MessageBox.Show("Sửa nhân viên thành công!");
                LamMoiForm();
            }
            catch (Exception ex)
            {
                MessageBox.Show("Không thể sửa nhân viên: " + ex.Message);
            }
        }

        // CAU 5: Xoa nhan vien duoc chon.
        private void btnXoa_Click(object sender, RoutedEventArgs e)
        {
            NHANVIEN dangChon = dgNhanVien.SelectedItem as NHANVIEN;
            if (dangChon == null)
            {
                MessageBox.Show("Vui lòng chọn nhân viên cần xóa.");
                return;
            }

            MessageBoxResult traLoi = MessageBox.Show(
                "Bạn có chắc muốn xóa nhân viên " + dangChon.HoTen + "?",
                "Xác nhận xóa",
                MessageBoxButton.YesNo,
                MessageBoxImage.Question);

            if (traLoi != MessageBoxResult.Yes) return;

            try
            {
                NHANVIEN nv = db.NHANVIENs.Find(dangChon.MaNV);
                if (nv == null)
                {
                    MessageBox.Show("Nhân viên không còn tồn tại.");
                    LamMoiForm();
                    return;
                }

                db.NHANVIENs.Remove(nv);
                db.SaveChanges();
                MessageBox.Show("Xóa nhân viên thành công!");
                LamMoiForm();
            }
            catch (Exception ex)
            {
                MessageBox.Show("Không thể xóa nhân viên: " + ex.Message);
            }
        }

using System;
using System.Data.Entity;
using System.Globalization;
using System.Linq;
using System.Windows;
using System.Windows.Controls;

namespace QUANLYSANPHAM
{
    /// <summary>
    /// Đề thi thử 3 - QUANLYSANPHAM (WPF .NET Framework, EF6 Database First).
    /// Entity Data Model cần có: QLSANPHAMEntities, SANPHAM, LOAISP.
    ///
    /// Tên control trong MainWindow.xaml:
    /// txtMaSP, txtTenSP, txtDonGia, txtSoLuong, cboLoaiSP, dgSanPham.
    ///
    /// Tên sự kiện cần khai báo trong XAML:
    /// dgSanPham_SelectionChanged, btnThem_Click, btnSua_Click,
    /// btnXoa_Click, btnLamMoi_Click.
    ///
    /// Giả định trong CSDL:
    /// SANPHAM (MaSP nvarchar, TenSP nvarchar, DonGia decimal(18,2),
    ///          SoLuong int, MaLoai nvarchar).
    /// LOAISP (MaLoai nvarchar, TenLoai nvarchar).
    /// Nếu tên DbContext/DbSet hoặc kiểu dữ liệu Entity tự sinh khác,
    /// cần thay tên/kiểu dữ liệu tương ứng.
    /// </summary>
    public partial class MainWindow : Window
    {
        private readonly QLSANPHAMEntities db = new QLSANPHAMEntities();

        public MainWindow()
        {
            InitializeComponent();
            TaiDuLieu();
        }

        // CÂU 3: Tải loại sản phẩm lên ComboBox và sản phẩm lên DataGrid.
        private void TaiDuLieu()
        {
            cboLoaiSP.ItemsSource = db.LOAISPs.ToList();
            cboLoaiSP.DisplayMemberPath = "TenLoai";
            cboLoaiSP.SelectedValuePath = "MaLoai";

            TaiDuLieuSanPham();
        }

        private void TaiDuLieuSanPham()
        {
            // Cột tên loại trong DataGrid có thể Binding="{Binding LOAISP.TenLoai}".
            // AsNoTracking lấy dữ liệu mới khi tải lại, Include lấy tên loại SP.
            dgSanPham.ItemsSource = db.SANPHAMs
                .AsNoTracking()
                .Include("LOAISP")
                .ToList();
        }

        // CÂU 4: Chọn dòng DataGrid -> đưa dữ liệu lên các control.
        private void dgSanPham_SelectionChanged(object sender, SelectionChangedEventArgs e)
        {
            SANPHAM sp = dgSanPham.SelectedItem as SANPHAM;
            if (sp == null)
                return;

            txtMaSP.Text = sp.MaSP;
            txtTenSP.Text = sp.TenSP;
            txtDonGia.Text = sp.DonGia.ToString(CultureInfo.CurrentCulture);
            txtSoLuong.Text = sp.SoLuong.ToString();
            cboLoaiSP.SelectedValue = sp.MaLoai;

            // Không sửa mã sản phẩm (khóa chính) khi đang chọn dòng.
            txtMaSP.IsReadOnly = true;
        }

        // Kiểm tra thông tin đầu vào và chuyển Đơn giá, Số lượng sang kiểu số.
        private bool KiemTraDuLieu(out decimal donGia, out int soLuong)
        {
            donGia = 0;
            soLuong = 0;

            if (string.IsNullOrWhiteSpace(txtMaSP.Text))
            {
                MessageBox.Show("Vui lòng nhập mã sản phẩm.");
                txtMaSP.Focus();
                return false;
            }

            if (string.IsNullOrWhiteSpace(txtTenSP.Text))
            {
                MessageBox.Show("Vui lòng nhập tên sản phẩm.");
                txtTenSP.Focus();
                return false;
            }

            if (!decimal.TryParse(txtDonGia.Text.Trim(),
                                  NumberStyles.Number,
                                  CultureInfo.CurrentCulture,
                                  out donGia) || donGia < 0)
            {
                MessageBox.Show("Đơn giá phải là số hợp lệ và không được âm.");
                txtDonGia.Focus();
                return false;
            }

            if (!int.TryParse(txtSoLuong.Text.Trim(), out soLuong) || soLuong < 0)
            {
                MessageBox.Show("Số lượng phải là số nguyên không âm.");
                txtSoLuong.Focus();
                return false;
            }

            if (cboLoaiSP.SelectedValue == null)
            {
                MessageBox.Show("Vui lòng chọn loại sản phẩm.");
                cboLoaiSP.Focus();
                return false;
            }

            return true;
        }

        // CÂU 5: Thêm sản phẩm mới.
        private void btnThem_Click(object sender, RoutedEventArgs e)
        {
            decimal donGia;
            int soLuong;
            if (!KiemTraDuLieu(out donGia, out soLuong))
                return;

            string maSP = txtMaSP.Text.Trim();
            if (db.SANPHAMs.Any(x => x.MaSP == maSP))
            {
                MessageBox.Show("Mã sản phẩm đã tồn tại.");
                return;
            }

            SANPHAM sp = new SANPHAM
            {
                MaSP = maSP,
                TenSP = txtTenSP.Text.Trim(),
                DonGia = donGia,
                SoLuong = soLuong,
                MaLoai = cboLoaiSP.SelectedValue.ToString()
            };

            try
            {
                db.SANPHAMs.Add(sp);
                db.SaveChanges();

                MessageBox.Show("Thêm sản phẩm thành công!");
                LamMoiForm();
            }
            catch (Exception ex)
            {
                // Bỏ đối tượng ở trạng thái Added nếu thêm thất bại.
                if (db.Entry(sp).State == EntityState.Added)
                    db.Entry(sp).State = EntityState.Detached;

                MessageBox.Show("Không thể thêm sản phẩm: " + ex.Message);
            }
        }

        // CÂU 5: Sửa sản phẩm đang chọn trong DataGrid.
        private void btnSua_Click(object sender, RoutedEventArgs e)
        {
            SANPHAM dangChon = dgSanPham.SelectedItem as SANPHAM;
            if (dangChon == null)
            {
                MessageBox.Show("Vui lòng chọn sản phẩm cần sửa.");
                return;
            }

            decimal donGia;
            int soLuong;
            if (!KiemTraDuLieu(out donGia, out soLuong))
                return;

            try
            {
                SANPHAM sp = db.SANPHAMs.Find(dangChon.MaSP);
                if (sp == null)
                {
                    MessageBox.Show("Sản phẩm không còn tồn tại.");
                    LamMoiForm();
                    return;
                }

                sp.TenSP = txtTenSP.Text.Trim();
                sp.DonGia = donGia;
                sp.SoLuong = soLuong;
                sp.MaLoai = cboLoaiSP.SelectedValue.ToString();

                db.SaveChanges();
                MessageBox.Show("Sửa sản phẩm thành công!");
                LamMoiForm();
            }
            catch (Exception ex)
            {
                MessageBox.Show("Không thể sửa sản phẩm: " + ex.Message);
            }
        }

        // CÂU 5: Xóa sản phẩm đang chọn trong DataGrid.
        private void btnXoa_Click(object sender, RoutedEventArgs e)
        {
            SANPHAM dangChon = dgSanPham.SelectedItem as SANPHAM;
            if (dangChon == null)
            {
                MessageBox.Show("Vui lòng chọn sản phẩm cần xóa.");
                return;
            }

            MessageBoxResult traLoi = MessageBox.Show(
                "Bạn có chắc muốn xóa sản phẩm '" + dangChon.TenSP + "'?",
                "Xác nhận xóa",
                MessageBoxButton.YesNo,
                MessageBoxImage.Question);

            if (traLoi != MessageBoxResult.Yes)
                return;

            try
            {
                SANPHAM sp = db.SANPHAMs.Find(dangChon.MaSP);
                if (sp == null)
                {
                    MessageBox.Show("Sản phẩm không còn tồn tại.");
                    LamMoiForm();
                    return;
                }

                db.SANPHAMs.Remove(sp);
                db.SaveChanges();

                MessageBox.Show("Xóa sản phẩm thành công!");
                LamMoiForm();
            }
            catch (Exception ex)
            {
                MessageBox.Show("Không thể xóa sản phẩm: " + ex.Message);
            }
        }

        // CÂU 4: Nhấn Làm mới để xóa các ô nhập và tải lại DataGrid.
        private void btnLamMoi_Click(object sender, RoutedEventArgs e)
        {
            LamMoiForm();
        }

        private void LamMoiForm()
        {
            dgSanPham.SelectedItem = null;

            txtMaSP.Clear();
            txtTenSP.Clear();
            txtDonGia.Clear();
            txtSoLuong.Clear();
            cboLoaiSP.SelectedIndex = -1;
            txtMaSP.IsReadOnly = false;

            TaiDuLieuSanPham();
            txtMaSP.Focus();
        }

        protected override void OnClosed(EventArgs e)
        {
            db.Dispose();
            base.OnClosed(e);
        }
    }
}
